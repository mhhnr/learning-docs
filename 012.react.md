## What is React?

React is a JavaScript library for building user interfaces, particularly web applications. Created by Facebook (now Meta), React focuses on creating reusable UI components and efficiently updating the user interface when data changes.

### Key Concepts:
- **Component-Based**: Build encapsulated components that manage their own state
- **Virtual DOM**: React creates a virtual representation of the DOM for efficient updates
- **Declarative**: Describe what the UI should look like, not how to achieve it
- **Unidirectional Data Flow**: Data flows down from parent to child components
- **JSX**: Write HTML-like syntax in JavaScript
- **Reusable**: Components can be reused throughout your application

### Why Use React?
```javascript
// Without React - Manual DOM manipulation
const button = document.getElementById('myButton');
const counter = document.getElementById('counter');
let count = 0;

button.addEventListener('click', () => {
    count++;
    counter.textContent = count;
});

// With React - Declarative approach
function Counter() {
    const [count, setCount] = useState(0);
    
    return (
        <div>
            <p>{count}</p>
            <button onClick={() => setCount(count + 1)}>
                Click me
            </button>
        </div>
    );
}
```

## JSX

JSX (JavaScript XML) is a syntax extension for JavaScript that looks similar to HTML. It's the way we write React components.

### Basic JSX
```javascript
// JSX looks like HTML but it's actually JavaScript
const element = <h1>Hello, World!</h1>;

// JSX with variables
const name = "Alice";
const greeting = <h1>Hello, {name}!</h1>;

// JSX with expressions
const user = { firstName: "John", lastName: "Doe" };
const element = (
  <h1>
    Hello, {user.firstName} {user.lastName}!
  </h1>
);

// JSX with attributes
const element = <img src="image.jpg" alt="Description" />;

// JSX with dynamic attributes
const imageUrl = "https://example.com/image.jpg";
const element = <img src={imageUrl} alt="Dynamic image" />;
```

### JSX Rules and Syntax
```javascript
// 1. Must return a single parent element
// ❌ This won't work
function Component() {
  return (
    <h1>Title</h1>
    <p>Paragraph</p>
  );
}

// ✅ Wrap in a parent element
function Component() {
  return (
    <div>
      <h1>Title</h1>
      <p>Paragraph</p>
    </div>
  );
}

// ✅ Or use React Fragment
function Component() {
  return (
    <React.Fragment>
      <h1>Title</h1>
      <p>Paragraph</p>
    </React.Fragment>
  );
}

// ✅ Or use short Fragment syntax
function Component() {
  return (
    <>
      <h1>Title</h1>
      <p>Paragraph</p>
    </>
  );
}

// 2. Use className instead of class
const element = <div className="my-class">Content</div>;

// 3. Use camelCase for attributes
const element = <input onChange={handleChange} autoComplete="off" />;

// 4. Self-closing tags must end with />
const element = <img src="image.jpg" alt="Description" />;
const element = <input type="text" />;

// 5. JavaScript expressions in curly braces
const isLoggedIn = true;
const element = (
  <div>
    {isLoggedIn ? <h1>Welcome back!</h1> : <h1>Please log in</h1>}
  </div>
);

// 6. Comments in JSX
const element = (
  <div>
    {/* This is a comment */}
    <h1>Hello World</h1>
  </div>
);
```

### Object and Array State
```javascript
// Object state
function UserProfile() {
  const [user, setUser] = useState({
    name: '',
    email: '',
    preferences: {
      theme: 'light',
      notifications: true
    }
  });

  // Update object state (shallow merge)
  const updateUser = (field, value) => {
    setUser(prevUser => ({
      ...prevUser,
      [field]: value
    }));
  };

  // Update nested object
  const updatePreference = (pref, value) => {
    setUser(prevUser => ({
      ...prevUser,
      preferences: {
        ...prevUser.preferences,
        [pref]: value
      }
    }));
  };

  return (
    <div>
      <input
        type="text"
        placeholder="Name"
        value={user.name}
        onChange={(e) => updateUser('name', e.target.value)}
      />
      <input
        type="email"
        placeholder="Email"
        value={user.email}
        onChange={(e) => updateUser('email', e.target.value)}
      />
      <label>
        <input
          type="checkbox"
          checked={user.preferences.notifications}
          onChange={(e) => updatePreference('notifications', e.target.checked)}
        />
        Enable notifications
      </label>
    </div>
  );
}

// Array state
function TodoList() {
  const [todos, setTodos] = useState([]);
  const [inputValue, setInputValue] = useState('');

  // Add todo
  const addTodo = () => {
    if (inputValue.trim()) {
      setTodos(prevTodos => [
        ...prevTodos,
        {
          id: Date.now(),
          text: inputValue,
          completed: false
        }
      ]);
      setInputValue('');
    }
  };

  // Toggle todo completion
  const toggleTodo = (id) => {
    setTodos(prevTodos =>
      prevTodos.map(todo =>
        todo.id === id ? { ...todo, completed: !todo.completed } : todo
      )
    );
  };

  // Remove todo
  const removeTodo = (id) => {
    setTodos(prevTodos => prevTodos.filter(todo => todo.id !== id));
  };

  return (
    <div>
      <div>
        <input
          type="text"
          value={inputValue}
          onChange={(e) => setInputValue(e.target.value)}
          onKeyPress={(e) => e.key === 'Enter' && addTodo()}
          placeholder="Add a todo"
        />
        <button onClick={addTodo}>Add</button>
      </div>
      
      <ul>
        {todos.map(todo => (
          <li key={todo.id}>
            <input
              type="checkbox"
              checked={todo.completed}
              onChange={() => toggleTodo(todo.id)}
            />
            <span style={{ 
              textDecoration: todo.completed ? 'line-through' : 'none' 
            }}>
              {todo.text}
            </span>
            <button onClick={() => removeTodo(todo.id)}>Delete</button>
          </li>
        ))}
      </ul>
    </div>
  );
}
```

### State Updates and Functional Updates
```javascript
function Counter() {
  const [count, setCount] = useState(0);

  // Regular state update
  const increment = () => {
    setCount(count + 1);
  };

  // Functional state update (recommended when depending on previous state)
  const incrementFunctional = () => {
    setCount(prevCount => prevCount + 1);
  };

  // Multiple rapid updates
  const incrementMultiple = () => {
    // This might not work as expected
    setCount(count + 1);
    setCount(count + 1);
    setCount(count + 1); // count might only increase by 1

    // This works correctly
    setCount(prevCount => prevCount + 1);
    setCount(prevCount => prevCount + 1);
    setCount(prevCount => prevCount + 1); // count increases by 3
  };

  return (
    <div>
      <p>Count: {count}</p>
      <button onClick={increment}>Increment</button>
      <button onClick={incrementFunctional}>Increment (Functional)</button>
      <button onClick={incrementMultiple}>Increment by 3</button>
    </div>
  );
}
```

function FormEvents() {

function useFormValidation(initialState, validate) {
  const [values, setValues] = useState(initialState);
  const [errors, setErrors] = useState({});
  const [isSubmitting, setIsSubmitting] = useState(false);

  const handleChange = (e) => {
    const { name, value } = e.target;
    setValues(prev => ({
      ...prev,
      [name]: value
    }));
  };

  const handleSubmit = (callback) => (e) => {
    e.preventDefault();
    const validationErrors = validate(values);
    setErrors(validationErrors);
    setIsSubmitting(true);

    if (Object.keys(validationErrors).length === 0) {
      callback();
    }
    setIsSubmitting(false);
  };

  return {
    values,
    errors,
    isSubmitting,
    handleChange,
    handleSubmit
  };
}

import React, { useState, useEffect } from 'react';

function EffectExamples() {
  const [count, setCount] = useState(0);
  const [data, setData] = useState(null);
  const [windowWidth, setWindowWidth] = useState(window.innerWidth);

  // Effect that runs after every render
  useEffect(() => {
    console.log('Component rendered');
  });

  // Effect that runs once (on mount)
  useEffect(() => {
    console.log('Component mounted');
    document.title = `React App`;
    
    // Cleanup function (runs on unmount)
    return () => {
      console.log('Component will unmount');
      document.title = 'React App'; // Reset title
    };
  }, []); // Empty dependency array

  // Effect that runs when count changes
  useEffect(() => {
    console.log(`Count changed to: ${count}`);
    document.title = `Count: ${count}`;
  }, [count]); // Dependency array with count

  // Effect with cleanup (event listeners)
  useEffect(() => {
    const handleResize = () => {
      setWindowWidth(window.innerWidth);
    };

    window.addEventListener('resize', handleResize);

    // Cleanup function
    return () => {
      window.removeEventListener('resize', handleResize);
    };
  }, []); // Run once on mount

  // Effect for data fetching
  useEffect(() => {
    let isCancelled = false;

    const fetchData = async () => {
      try {
        console.log('Fetching data...');
        // Simulate API call
        const response = await fetch('https://jsonplaceholder.typicode.com/posts/1');
        const result = await response.json();
        
        // Only update state if component is still mounted
        if (!isCancelled) {
          setData(result);
        }
      } catch (error) {
        if (!isCancelled) {
          console.error('Error fetching data:', error);
        }
      }
    };

    fetchData();

    // Cleanup function to cancel the request
    return () => {
      isCancelled = true;
    };
  }, []); // Run once on mount

  // Effect with multiple dependencies
  useEffect(() => {
    if (count > 0 && data) {
      console.log('Both count and data are available');
    }
  }, [count, data]); // Runs when count OR data changes

  return (
    <div>
      <h2>Effect Examples</h2>
      
      <div>
        <p>Count: {count}</p>
        <button onClick={() => setCount(count + 1)}>Increment</button>
      </div>

      <div>
        <p>Window width: {windowWidth}px</p>
        <p>Resize the window to see the effect!</p>
      </div>

      <div>
        <h3>Fetched Data:</h3>
        {data ? (
          <div>
            <p><strong>Title:</strong> {data.title}</p>
            <p><strong>Body:</strong> {data.body}</p>
          </div>
        ) : (
          <p>Loading...</p>

    </div>
  );

### useContext Hook
```javascript
import React, { createContext, useContext, useState } from 'react';

// Create contexts
const ThemeContext = createContext();
const UserContext = createContext();

// Theme provider component
function ThemeProvider({ children }) {
  const [theme, setTheme] = useState('light');

  const toggleTheme = () => {
    setTheme(prevTheme => prevTheme === 'light' ? 'dark' : 'light');
  };

  const value = {
    theme,
    toggleTheme
  };

  return (
    <ThemeContext.Provider value={value}>
      {children}
    </ThemeContext.Provider>
  );
}

// User provider component
function UserProvider({ children }) {
  const [user, setUser] = useState(null);

  const login = (userData) => {
    setUser(userData);
  };

  const logout = () => {
    setUser(null);
  };

  const value = {
    user,
    login,
    logout,
    isLoggedIn: !!user
  };

  return (
    <UserContext.Provider value={value}>
      {children}
    </UserContext.Provider>
  );
}

// Custom hooks to use contexts
function useTheme() {
  const context = useContext(ThemeContext);
  if (!context) {
    throw new Error('useTheme must be used within a ThemeProvider');
  }
  return context;
}

function useUser() {
  const context = useContext(UserContext);
  if (!context) {
    throw new Error('useUser must be used within a UserProvider');
  }
  return context;
}
