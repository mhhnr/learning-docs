## 1. Introduction to TypeScript

TypeScript is a superset of JavaScript that adds **optional static typing** to the language. It compiles down to plain JavaScript, meaning any valid JavaScript code is also valid TypeScript code. Developed by Microsoft, TypeScript aims to make JavaScript development more scalable and maintainable, especially for large applications.

### Key Features:
-   **Static Typing**: Allows you to define types for variables, function parameters, and return values, catching errors at compile-time rather than runtime.
-   **Compilation**: TypeScript code (`.ts` or `.tsx` files) is compiled into JavaScript (`.js` files) that browsers and Node.js can understand.
-   **Superset of JavaScript**: All valid JavaScript code is valid TypeScript. This means a gradual adoption is possible.
-   **Rich Type System**: Includes interfaces, enums, generics, union types, intersection types, and more.
-   **Improved Tooling**: Provides excellent IDE support (autocompletion, refactoring, error checking) thanks to its type information.
-   **Object-Oriented Features**: Enhances classes, interfaces, and modules.

```typescript
// Your first TypeScript program
let message: string = "Hello, TypeScript!"; // Type annotation for 'message'
console.log(message); // This prints "Hello, TypeScript!" to the console

// message = 123; // Error: Type 'number' is not assignable to type 'string'.
```

## 3. Variables and Data Types (and Type System Basics)

TypeScript introduces the concept of **types** to variables, allowing for better code reliability and tooling.

### Variable Declarations
```typescript
// var (function-scoped, can be redeclared, generally avoided in modern TS/JS)
var oldStyle: string = "I'm var";
var oldStyle: string = "I can be redeclared";  // No error in JS, TS will warn if strict is on

// let (block-scoped, cannot be redeclared in same scope)
let modernStyle: string = "I'm let";
// let modernStyle: string = "Error!";  // SyntaxError: Cannot redeclare block-scoped variable 'modernStyle'.

// const (block-scoped, cannot be reassigned)
const constant: string = "I cannot be reassigned";
// constant = "Error!";  // TypeError: Cannot assign to 'constant' because it is a constant.

// Type Inference: TypeScript often figures out the type for you
let inferredString = "Hello!"; // inferredString is type 'string'
// inferredString = 123; // Error: Type 'number' is not assignable to type 'string'.

// Variable naming rules:
// - Must start with letter, underscore, or dollar sign
// - Can contain letters, numbers, underscores, dollar signs
// - Case-sensitive
let validName: number = 1;
let _underscore: number = 2;
let $dollar: number = 3;
let camelCase: number = 4;
```

### Primitive Data Types

TypeScript provides explicit types for JavaScript's primitives.

#### Numbers (`number`)
```typescript
// All numbers are floating-point (like JS)
let integer: number = 42;
let decimal: number = 3.14159;
let negative: number = -7;
let scientific: number = 1.23e-4;  // Scientific notation

// Special numeric values
let infinity: number = Infinity;
let negInfinity: number = -Infinity;
let notANumber: number = NaN;

// Number operations - types are checked
console.log(5 / 0);        // Infinity
console.log(-5 / 0);       // -Infinity
// console.log("hello" * 5);  // Error: The right-hand side of an arithmetic operation must be of type 'any', 'number', 'bigint' or an enum type.

// Checking for NaN
console.log(isNaN(NaN));           // true
console.log(Number.isNaN(NaN));    // true (preferred)
```

#### Strings (`string`)

```typescript
let doubleQuoted: string = "World";

```

#### Booleans (`boolean`)
```typescript
// Boolean values
let isTrue: boolean = true;
let isFalse: boolean = false;

// Boolean conversion (truthy/falsy values)
console.log(Boolean(1));        // true
console.log(Boolean(0));        // false
console.log(Boolean(""));       // false (empty string)
console.log(Boolean("hello"));  // true
console.log(Boolean(null));     // false
console.log(Boolean(undefined)); // false

console.log(Boolean({}));       // true (empty object)
```

#### `null` and `undefined` (`null`, `undefined`)
```typescript
// undefined: variable declared but not assigned
let notAssigned: undefined; // Explicit type annotation
console.log(notAssigned);  // undefined
console.log(typeof notAssigned);  // "undefined"

// null: intentional absence of value
let intentionallyEmpty: null = null; // Explicit type annotation
console.log(intentionallyEmpty);  // null
console.log(typeof intentionallyEmpty);  // "object" (this is a known JS quirk, TS acknowledges it)

// When strictNullChecks is true (recommended in tsconfig.json):
// You cannot assign null to a variable typed as string, number, boolean, etc.
// let nameStrict: string = null; // Error: Type 'null' is not assignable to type 'string'.

// Union types are used for nullable values
let nullableName: string | null = "John";
nullableName = null; // OK
```

#### Symbols (`symbol`, ES6)
```typescript
// Symbols are unique identifiers
let sym1: symbol = Symbol();
let sym2: symbol = Symbol();
console.log(sym1 === sym2);  // false (each symbol is unique)

let sym3: symbol = Symbol("description");
console.log(sym3.toString());  // "Symbol(description)"

// Symbols as object properties (string index signature won't see them)
let objWithSymbol: { [key: string]: string; [sym1]: string; } = {}; // Explicitly define symbol index signature
objWithSymbol[sym1] = "value1";
objWithSymbol[sym2] = "value2"; // This would create a new symbol property dynamically
console.log(objWithSymbol[sym1]);  // "value1"
```

#### BigInt (`bigint`, ES2020)
```typescript
// For integers larger than Number.MAX_SAFE_INTEGER
let bigNumber: bigint = 1234567890123456789012345678901234567890n;
let anotherBig: bigint = BigInt("1234567890123456789012345678901234567890");

console.log(typeof bigNumber);  // "bigint"
console.log(bigNumber + 1n);    // Need 'n' suffix for BigInt operations
// console.log(bigNumber + 1); // Error: Operator '+' cannot be applied to types 'bigint' and 'number'.
```

### Special Types in TypeScript

#### `any` (Avoid if possible)
```typescript
// 'any' disables type checking for this variable. Use with caution.
let anyValue: any = "hello";
anyValue = 10;
anyValue = true;
anyValue.doSomething(); // No compile-time error, but might fail at runtime
```

#### `unknown` (Safer alternative to `any`)
```typescript
// 'unknown' means we don't know the type, but TypeScript forces you to narrow it down.
let unknownValue: unknown = "hello";
unknownValue = 10;

// console.log(unknownValue.length); // Error: Object is of type 'unknown'.
if (typeof unknownValue === 'string') {
    console.log(unknownValue.length); // OK, type is narrowed to 'string'
}
```

#### `void`
```typescript
// 'void' is used for functions that do not return any value.
function logMessage(message: string): void {
    console.log(message);
}

let result: void = logMessage("Done"); // result will be undefined
```

#### `never`
```typescript
// 'never' represents values that never occur. Used for functions that always throw an error
// or functions that never return (e.g., infinite loops).
function error(message: string): never {
    throw new Error(message);
}

function infiniteLoop(): never {
    while (true) { /* ... */ }
}

// Function whose return type is inferred as 'never' because it's unreachable
function processInput(input: string | number): string {
    if (typeof input === 'string') {
        return `String: ${input}`;
    } else if (typeof input === 'number') {
        return `Number: ${input}`;
    }
    // const check: never = input; // 'input' here is of type 'never' if all branches handled 'string' | 'number'
    // This helps ensure all possible cases are handled in discriminated unions.
    error("Should not reach here");
}
```

### Type Checking
```typescript
// typeof operator (runtime JS operator)
console.log(typeof 42);         // "number"
console.log(typeof "hello");    // "string"
console.log(typeof true);       // "boolean"
console.log(typeof undefined);  // "undefined"
console.log(typeof null);       // "object" (JS quirk!)
console.log(typeof {});         // "object"
console.log(typeof []);         // "object"
console.log(typeof function(){}); // "function"

// More precise type checking (runtime JS checks)
console.log(Array.isArray([]));     // true
console.log(Array.isArray({}));     // false
console.log(null === null);         // true

// TypeScript's static type checking is done at compile time.
// E.g., The compiler will tell you if you try to assign a string to a number variable.
// let num: number = "five"; // Error: Type 'string' is not assignable to type 'number'.
```

## 4. Operators

Operators in TypeScript work identically to JavaScript as they operate on values. TypeScript's role is to ensure that the types involved in operations are compatible.

### Arithmetic Operators
```typescript
let a: number = 10, b: number = 3;

console.log(a + b);   // Addition: 13
console.log(a - b);   // Subtraction: 7
console.log(a * b);   // Multiplication: 30
console.log(a / b);   // Division: 3.3333...
console.log(a % b);   // Modulus (remainder): 1
console.log(a ** b);  // Exponentiation (ES2016): 1000

// Unary operators
console.log(+a);      // Unary plus: 10
console.log(-a);      // Unary minus: -10
console.log(++a);     // Pre-increment: 11 (a becomes 11)
console.log(a++);     // Post-increment: 11 (then a becomes 12)
console.log(--a);     // Pre-decrement: 11 (a becomes 11)
console.log(a--);     // Post-decrement: 11 (then a becomes 10)

// TypeScript ensures types are compatible
// let result = "hello" + 5; // Valid (string concatenation)
// let result2 = "hello" * 5; // Error: Operator '*' cannot be applied to types 'string' and 'number'.
```

### Assignment Operators
```typescript
let x: number = 10;

x += 5;   // x = x + 5;  (15)
x -= 3;   // x = x - 3;  (12)
x *= 2;   // x = x * 2;  (24)
x /= 4;   // x = x / 4;  (6)
x %= 5;   // x = x % 5;  (1)
x **= 3;  // x = x ** 3; (1)

// Logical assignment (ES2021) - TypeScript correctly infers and checks types here
let y: number | false = false;
y ||= 10;  // y = y || 10; (assigns if y is falsy)
console.log(y); // 10

let z: number | string = "hello";
z &&= 5;   // z = z && 5;  (assigns if z is truthy)
console.log(z); // 5

let k: number | null | undefined = null;
k ??= 7;   // k = k ?? 7;  (assigns if k is null/undefined)
console.log(k); // 7
```

### Comparison Operators
```typescript
let a: number = 10;
let b: string = "10";
let c: number = 5;

// Equality (loose - performs type coercion in JS, TS will usually warn if types are different)
console.log(a == b);   // true (10 == "10") - TS might warn under strict equality rules or type coercion.
console.log(a != c);   // true (10 != 5)

// Strict equality (no type coercion) - Recommended in TypeScript
console.log(a === b);  // false (10 !== "10") - TS strongly encourages this
console.log(a !== b);  // true (different types)

// Relational operators
console.log(a > c);    // true
console.log(a < c);    // false
console.log(a >= 10);  // true
console.log(c <= 5);   // true
```

### Logical Operators
```typescript
let x: boolean = true;
let y: boolean = false;

// Logical AND
console.log(x && y);        // false
console.log(true && true);  // true

// Logical OR
console.log(x || y);         // true
console.log(false || false); // false

// Logical NOT
console.log(!x);  // false
console.log(!y);  // true

// Short-circuit evaluation - TypeScript handles return types correctly
function someFunction(): number {
    console.log("someFunction called");
    return 100;
}

let result: number | boolean = x && someFunction(); // someFunction only called if x is true
console.log(result); // 100

let value: string | number = y || "default";       // "default" used if y is falsy
console.log(value); // "default"
```

### Bitwise Operators
```typescript
let a: number = 5;   // Binary: 101
let b: number = 3;   // Binary: 011

console.log(a & b);   // Bitwise AND: 1 (001)
console.log(a | b);   // Bitwise OR: 7 (111)
console.log(a ^ b);   // Bitwise XOR: 6 (110)
console.log(~a);      // Bitwise NOT: -6
console.log(a << 1);  // Left shift: 10 (1010)
console.log(a >> 1);  // Right shift: 2 (010)
console.log(a >>> 1); // Unsigned right shift: 2
```

### Other Operators
```typescript
// Ternary operator - TypeScript correctly infers the union return type
let age: number = 18;
let statusText: string = age >= 18 ? "adult" : "minor";
console.log(statusText);  // "adult"

// Nullish coalescing operator (ES2020) - TypeScript recognizes null/undefined checks
let userName: string | null = null;
let defaultName: string = userName ?? "Anonymous";
console.log(defaultName);  // "Anonymous"

// Optional chaining operator (ES2020) - TypeScript infers optional properties
interface UserProfile {
    name: string;
    age?: number;
}

interface User {
    profile?: UserProfile;
}

let user: User = { profile: { name: "John" } };
console.log(user?.profile?.name);     // "John"
console.log(user?.profile?.age);      // undefined
console.log(user?.nonexistent?.name); // undefined (no error)

// typeof operator (runtime JS operator)
console.log(typeof "hello");  // "string"
console.log(typeof 42);       // "number"

// instanceof operator (runtime JS operator, also a type guard in TS)
let myArray: number[] = [];
console.log(myArray instanceof Array);   // true
console.log({} instanceof Object);  // true

// in operator (runtime JS operator, also a type guard in TS)
let objProp: { name: string; age: number; } = { name: "John", age: 30 };
console.log("name" in objProp);     // true
console.log("address" in objProp);  // false
```

## 5. Control Flow

Control flow statements are identical to JavaScript. TypeScript's contribution here is ensuring type safety within these blocks and providing **control flow analysis** to narrow down types.

### Conditional Statements
```typescript
// if statement - TypeScript narrows types inside blocks
let score: number = 85;
if (score >= 90) {
    console.log("Grade: A");
} else if (score >= 80) {
    console.log("Grade: B");
} else if (score >= 70) {
    console.log("Grade: C");
} else {
    console.log("Grade: F");
}

// Type narrowing with if
let mixedValue: string | number = "hello";
if (typeof mixedValue === 'string') {
    console.log(mixedValue.length); // mixedValue is narrowed to 'string' here
} else {
    console.log(mixedValue * 2);    // mixedValue is narrowed to 'number' here
}

// Ternary operator
let gradeStatus: string = score >= 60 ? "Pass" : "Fail";

// switch statement
let day: string = "Monday";
switch (day) {
    case "Monday":
        console.log("Start of work week");
        break;
    case "Friday":
        console.log("TGIF!");
        break;
    case "Saturday":
    case "Sunday":
        console.log("Weekend!");
        break;
    default:
        console.log("Midweek");
}
```

### Loops

#### `for` Loop
```typescript
// Traditional for loop
for (let i: number = 0; i < 5; i++) {
    console.log(i);  // Prints 0, 1, 2, 3, 4
}

// for...in loop (iterates over object properties - warns if not explicitly typed)
let personObj: { name: string; age: number; job: string; } = { name: "John", age: 30, job: "developer" };
for (let key in personObj) {
    // TypeScript might warn that 'key' is string, and personObj[key] could be any type.
    // Use type assertion or check:
    console.log(`${key}: ${personObj[key as keyof typeof personObj]}`);
}

// for...of loop (iterates over iterable values - type-safe)
let fruits: string[] = ["apple", "banana", "cherry"];
for (let fruit of fruits) {
    console.log(fruit); // fruit is inferred as 'string'
}

// for...of with index (using .entries())
for (let [index, fruit] of fruits.entries()) {
    console.log(`${index}: ${fruit}`);
}
```

#### `while` Loop
```typescript
// while loop
let count: number = 0;
while (count < 5) {
    console.log(count);
    count++;
}

// do...while loop (executes at least once)
let xLoop: number = 0;
do {
    console.log(xLoop);
    xLoop++;
} while (xLoop < 3);
```

### Loop Control
```typescript
// break statement
for (let i: number = 0; i < 10; i++) {
    if (i === 5) {
        break;  // Exit the loop
    }
    console.log(i);  // Prints 0, 1, 2, 3, 4
}

// continue statement
for (let i: number = 0; i < 5; i++) {
    if (i === 2) {
        continue;  // Skip the rest of this iteration
    }
    console.log(i);  // Prints 0, 1, 3, 4
}

// Labels (rarely used)
outer: for (let i: number = 0; i < 3; i++) {
    for (let j: number = 0; j < 3; j++) {
        if (i === 1 && j === 1) {
            break outer;  // Break out of both loops
        }
        console.log(`${i}, ${j}`);
    }
}
```

## 6. Functions

TypeScript significantly enhances functions by allowing explicit type annotations for parameters and return values, ensuring correct usage and better documentation.

### Function Declarations
```typescript
// Function declaration (hoisted) with type annotations
function greet(name: string): string {
    return `Hello, ${name}!`;
}

console.log(greet("Alice"));  // "Hello, Alice!"
// console.log(greet(123)); // Error: Argument of type 'number' is not assignable to parameter of type 'string'.

// Function expression with type annotations
const greetExpression = function(name: string): string {
    return `Hello, ${name}!`;
};

// Arrow functions (ES6) with type annotations
const greetArrow = (name: string): string => {
    return `Hello, ${name}!`;
};

// Concise arrow function (single expression) with inferred return type
const greetConcise = (name: string): string => `Hello, ${name}!`;

// Arrow function with multiple parameters
const add = (a: number, b: number): number => a + b;

// Arrow function with no parameters
const sayHello = (): string => "Hello!";

// Void return type
function logMessage(message: string): void {
    console.log(message);
}
```

### Parameters and Arguments
```typescript
// Default parameters (ES6) with types
function greetWithDefaults(name: string = "World", greeting: string = "Hello"): string {
    return `${greeting}, ${name}!`;
}

console.log(greetWithDefaults());               // "Hello, World!"
console.log(greetWithDefaults("Alice"));        // "Hello, Alice!"
console.log(greetWithDefaults("Bob", "Hi"));    // "Hi, Bob!"

// Optional parameters (using ?)
function displayInfo(name: string, age?: number): string {
    return `Name: ${name}` + (age !== undefined ? `, Age: ${age}` : "");
}

console.log(displayInfo("John"));      // "Name: John"
console.log(displayInfo("Jane", 30));  // "Name: Jane, Age: 30"

// Rest parameters (variable number of arguments) with array type
function sum(...numbers: number[]): number {
    return numbers.reduce((total, num) => total + num, 0);
}

console.log(sum(1, 2, 3, 4));  // 10

// Spread operator in function calls (JS feature, TS checks array types)
let numbersArray: number[] = [1, 2, 3, 4, 5];
console.log(Math.max(...numbersArray));  // 5

// Destructuring parameters with type annotations
interface PersonInfo {
    name: string;
    age: number;
    job?: string; // Optional property
}

function describePerson({ name, age, job = "unemployed" }: PersonInfo): string {
    return `${name} is ${age} years old and works as a ${job}.`;
}

let personObject: PersonInfo = { name: "Alice", age: 30, job: "developer" };
console.log(describePerson(personObject));

// Function Overloads
// Multiple signatures for a single function implementation
function combine(a: string, b: string): string;
function combine(a: number, b: number): number;
function combine(a: any, b: any): any { // Actual implementation (often uses 'any')
    if (typeof a === 'string' && typeof b === 'string') {
        return a + b;
    }
    if (typeof a === 'number' && typeof b === 'number') {
        return a + b;
    }
    throw new Error("Invalid arguments");
}

console.log(combine("Hello", " World")); // "Hello World" (string signature)
console.log(combine(10, 20));          // 30 (number signature)
// console.log(combine("hello", 10)); // Error: No overload matches this call.
```

### Function Scope and Closures
```typescript
// Global scope
let globalVar: string = "I'm global";

function outerFunction(x: number) {
    // Function scope
    let outerVar: string = "I'm in outer function";
    
    function innerFunction(y: number): number {
        // Inner function has access to outer variables (closure)
        let innerVar: string = "I'm in inner function";
        console.log(globalVar);  // Accessible
        console.log(outerVar);   // Accessible
        console.log(innerVar);   // Accessible
        return x + y;
    }
    
    return innerFunction;
}

// Closure example - types enhance clarity
function createCounter(): { increment: () => number; decrement: () => number; getCount: () => number; } {
    let count: number = 0;
    return {
        increment: () => {
            count++;
            return count;
        },
        decrement: () => {
            count--;
            return count;
        },
        getCount: () => count
    };
}

let counter1 = createCounter();
let counter2 = createCounter();

console.log(counter1.increment());  // 1
console.log(counter1.increment());  // 2
console.log(counter2.increment());  // 1 (independent counter)
console.log(counter1.getCount());   // 2
```

### Higher-Order Functions
```typescript
// Functions that take other functions as arguments (type definitions for callbacks)
type OperationFn = (a: number, b: number) => number;

function calculate(operation: OperationFn, a: number, b: number): number {
    return operation(a, b);
}

const addOperation: OperationFn = (x, y) => x + y;
const multiplyOperation: OperationFn = (x, y) => x * y;

console.log(calculate(addOperation, 5, 3));      // 8
console.log(calculate(multiplyOperation, 4, 6)); // 24

// Functions that return other functions (closure, type definitions for returned function)
type MultiplierFn = (x: number) => number;

function createMultiplier(multiplier: number): MultiplierFn {
    return function(x: number): number {
        return x * multiplier;
    };
}

const double: MultiplierFn = createMultiplier(2);
const triple: MultiplierFn = createMultiplier(3);

console.log(double(5));  // 10
console.log(triple(4));  // 12
```

### Recursion
```typescript
// Factorial function with type annotations
function factorial(n: number): number {
    if (n <= 1) {
        return 1;  // Base case
    }
    return n * factorial(n - 1);  // Recursive case
}

console.log(factorial(5));  // 120

// Fibonacci sequence with type annotations
```

## 7. Interfaces and Type Aliases

These are fundamental TypeScript features for defining the shape of objects and functions.

### Interfaces
Interfaces define contracts for the shape of objects. They are purely compile-time constructs and are removed during compilation to JavaScript.

```typescript
// Basic Interface
interface Person {
    name: string;
    age: number;
    job?: string; // Optional property
    readonly id: string; // Readonly property
}

// Using the interface
let user1: Person = {
    name: "Alice",
    age: 30,
    id: "abc-123"
};

let user2: Person = {
    name: "Bob",
    age: 25,
    job: "developer",
    id: "def-456"
};

// user1.id = "new-id"; // Error: Cannot assign to 'id' because it is a read-only property.

// Interface with method signature
interface Greetable {
    name: string;
    greet(phrase: string): void;
}

class Greeter implements Greetable {
    name: string;
    constructor(n: string) {
        this.name = n;
    }
    greet(phrase: string) {
        console.log(`${phrase} ${this.name}`);
    }
}

let myGreeter = new Greeter("Charlie");
myGreeter.greet("Hi there,");

// Extending Interfaces
interface AgedPerson extends Person {
    yearsActive: number;
}

let seniorUser: AgedPerson = {
    name: "David",
    age: 60,
    job: "consultant",
    id: "xyz-789",
    yearsActive: 35
};

// Interface for function types
interface SumFunction {
    (a: number, b: number): number;
}

const mySum: SumFunction = (x, y) => x + y;
console.log(mySum(7, 3)); // 10

// Interface for indexable types (e.g., arrays or dictionaries)
interface StringArray {
    [index: number]: string;
}

let myArray: StringArray = ["hello", "world"];

interface StringDictionary {
    [key: string]: string;
}

let myDictionary: StringDictionary = {
    greeting: "Hello",
    farewell: "Goodbye"
};
```

### Type Aliases
Type aliases create a new name for a type. They don't create a new type, just a new name for an existing one. They are more flexible than interfaces for certain scenarios, like union types.

```typescript
// Type alias for a primitive
type UserId = string;
let currentUserId: UserId = "user-123";

// Type alias for an object shape (similar to interface)
type Product = {
    name: string;
    price: number;
    inStock: boolean;
};

let laptop: Product = {
    name: "Laptop Pro",
    price: 1200,
    inStock: true
};

// Type alias for a function type
type MathOperation = (a: number, b: number) => number;
const divideOperation: MathOperation = (x, y) => x / y;
console.log(divideOperation(10, 2)); // 5

// Union Types (cannot be done with interfaces)
type Status = "pending" | "success" | "error";
let requestStatus: Status = "success";
// requestStatus = "failed"; // Error: Type '"failed"' is not assignable to type 'Status'.

type ID = number | string;
let userId: ID = 123;
userId = "abc";

// Intersection Types (combining multiple types)
interface HasName {
    name: string;
}

interface HasAge {
    age: number;
}

type NamedAndAged = HasName & HasAge;

let personProfile: NamedAndAged = {
    name: "Eve",
    age: 40
};

// Type alias for a tuple
type RGB = [number, number, number];
let color: RGB = [255, 0, 0];
// let invalidColor: RGB = [255, 0]; // Error: Source has 2 elements, but target requires 3.

// Interfaces vs Type Aliases:
// - Interfaces can be 'implemented' by classes (implements keyword).
// - Interfaces can be 'merged' (declaration merging) if declared multiple times.
// - Type aliases are more flexible for unions, intersections, and primitive type renaming.
// - For defining object shapes, both are often interchangeable, but interfaces are generally preferred for clarity and extendability.
```

TypeScript enforces the structure and types within objects and arrays.

### Objects
```typescript

```

### Object Methods and Properties (Runtime JS features)
```typescript
let personObjMethods: { name: string; age: number; job: string; } = { name: "Alice", age: 25, job: "designer" };

// Object.keys() - get all property names (returns string[])
console.log(Object.keys(personObjMethods));  // ["name", "age", "job"]

// Object.values() - get all values (returns (string | number)[])
console.log(Object.values(personObjMethods));  // ["Alice", 25, "designer"]

// Object.entries() - get key-value pairs (returns [string, string | number][])
console.log(Object.entries(personObjMethods));  // [["name", "Alice"], ["age", 25], ["job", "designer"]]

// Object.assign() - copy properties (TS correctly infers the resulting type)
let copy: typeof personObjMethods = Object.assign({}, personObjMethods);
let extended: UserProfile = Object.assign({}, personObjMethods, { salary: 60000 });

// Object.freeze() - make object immutable (TS can't enforce immutability strictly after freeze, runtime)
Object.freeze(personObjMethods);
// personObjMethods.age = 26;  // Will fail at runtime in strict mode, TS won't necessarily error here unless you use a Readonly type.

// Object.seal() - prevent adding/deleting properties
let sealedObj: { a: number; b: number; } = { a: 1, b: 2 };
Object.seal(sealedObj);
sealedObj.a = 10;  // Can modify existing properties
// sealedObj.c = 3;  // Can't add new properties at runtime

// hasOwnProperty() - check if property exists
console.log(personObjMethods.hasOwnProperty("name"));  // true
console.log(personObjMethods.hasOwnProperty("toString"));  // false (inherited)
```

### Arrays
TypeScript arrays are typed, meaning you specify the type of elements they can contain.

```typescript
// Array creation with type annotations
let emptyArray: any[] = []; // Array can contain any type
let numbers: number[] = [1, 2, 3, 4, 5]; // Array of numbers
let mixed: (number | string | boolean | null | { name: string })[] = [1, "hello", true, null, { name: "John" }]; // Array with union types
let arrayFromRange: number[] = Array.from({ length: 5 }, (_, i) => i + 1);  // [1, 2, 3, 4, 5]

// Generic Array type syntax
let names: Array<string> = ["Alice", "Bob"];

// Tuple types (fixed size, fixed types at specific positions)
let personTuple: [string, number, boolean] = ["Alice", 30, true];
// personTuple[0] = 123; // Error: Type 'number' is not assignable to type 'string'.
// personTuple[3] = "extra"; // Error: Tuple type '[string, number, boolean]' of length '3' has no element at index '3'.

// Accessing elements
console.log(numbers[0]);    // First element: 1
// console.log(numbers[-1]);   // No negative indexing, TS warns about out-of-bounds access
console.log(numbers[numbers.length - 1]);  // Last element: 5

// Array properties
console.log(numbers.length);  // 5

// Modifying arrays (TS ensures type compatibility)
numbers.push(6);           // Add to end: [1, 2, 3, 4, 5, 6]
numbers.pop();             // Remove from end: [1, 2, 3, 4, 5]
numbers.unshift(0);        // Add to beginning: [0, 1, 2, 3, 4, 5]
numbers.shift();           // Remove from beginning: [1, 2, 3, 4, 5]

// Splice method (add/remove elements at any position)
numbers.splice(2, 1);      // Remove 1 element at index 2: [1, 2, 4, 5]
numbers.splice(2, 0, 3);   // Insert 3 at index 2: [1, 2, 3, 4, 5]
// numbers.splice(1, 2, "a", "b");  // Error if array is number[], types must match: [1, "a", "b", 4, 5]
let mixedArray: (number | string)[] = [1, 2, 3, 4, 5];
mixedArray.splice(1, 2, "a", "b"); // OK if types are compatible
```

### Array Methods (Typescript provides excellent type inference for these)
```typescript
let numbersArr: number[] = [1, 2, 3, 4, 5];
let wordsArr: string[] = ["apple", "banana", "cherry"];

// Iteration methods
numbersArr.forEach((num: number, index: number) => {
    console.log(`${index}: ${num}`);
});

// Map - transform each element (result type is inferred)
let doubled: number[] = numbersArr.map(num => num * 2);  // [2, 4, 6, 8, 10]
let numToString: string[] = numbersArr.map(num => num.toString()); // ["1", "2", "3", "4", "5"]

// Filter - select elements that meet condition (result type same as original)
let evens: number[] = numbersArr.filter(num => num % 2 === 0);  // [2, 4]

// Reduce - accumulate values (result type is inferred, or specified with generic)
let sumReduced: number = numbersArr.reduce((total: number, num: number) => total + num, 0);  // 15
let productReduced: number = numbersArr.reduce((total: number, num: number) => total * num, 1);  // 120

// Find - get first element that meets condition (result type is element type or undefined)
let found: number | undefined = numbersArr.find(num => num > 3);  // 4

// Some - check if any element meets condition (returns boolean)
let hasEven: boolean = numbersArr.some(num => num % 2 === 0);  // true

// Every - check if all elements meet condition (returns boolean)
let allPositive: boolean = numbersArr.every(num => num > 0);  // true

// Sort (can require a type-safe comparison function)
let sortedWords: string[] = wordsArr.sort();  // ["apple", "banana", "cherry"]
let sortedNumbers: number[] = [...numbersArr].sort((a, b) => b - a);  // [5, 4, 3, 2, 1] (descending)

// Join - convert array to string
let joined: string = wordsArr.join(", ");  // "apple, banana, cherry"

// Slice - extract portion of array (doesn't modify original, returns same type)
let portion: number[] = numbersArr.slice(1, 4);  // [2, 3, 4]

// Concat - combine arrays (returns combined type)
let combined: number[] = numbersArr.concat([6, 7, 8]);  // [1, 2, 3, 4, 5, 6, 7, 8]

// Includes - check if array contains element
let hasThree: boolean = numbersArr.includes(3);  // true

// IndexOf - find index of element
let index: number = numbersArr.indexOf(3);  // 2

// Reverse - reverse array in place
numbersArr.reverse();  // [5, 4, 3, 2, 1]
```

### Array Destructuring (ES6)

```typescript
console.log(y);  // 0 (default value)
```

## 9. Enums

Enums are a powerful TypeScript feature that allows you to define a set of named constants. They make your code more readable and prevent typos.

```typescript
// Numeric Enum (default: starts at 0)
enum Direction {
    Up,      // 0
    Down,    // 1
    Left,    // 2
    Right    // 3
}

let playerDirection: Direction = Direction.Up;
console.log(playerDirection);        // 0
console.log(Direction.Right);      // 3
console.log(Direction[0]);         // "Up" (reverse mapping)

// Numeric Enum with custom starting value
enum StatusCode {
    Success = 200,
    NotFound = 404,
    ServerError = 500
}

let responseStatus: StatusCode = StatusCode.Success;
console.log(responseStatus);        // 200
console.log(StatusCode.NotFound);   // 404

// String Enum (no reverse mapping, better for debugging)
enum LogLevel {
    INFO = "INFO",
    WARN = "WARN",
    ERROR = "ERROR",
    DEBUG = "DEBUG"
}

let currentLogLevel: LogLevel = LogLevel.WARN;
console.log(currentLogLevel);       // "WARN"
// console.log(LogLevel["INFO"]); // "INFO"
// console.log(LogLevel[0]); // Error: Element implicitly has an 'any' type because expression of type '0' can't be used to index type 'typeof LogLevel'.

function log(level: LogLevel, message: string): void {
    console.log(`[${level}] ${message}`);
}
log(LogLevel.INFO, "User logged in.");

// Heterogeneous Enum (mix of numeric and string, not recommended)
enum Mixed {
    No = 0,
    Yes = "YES"
}

// Const Enums (purely compile-time, no generated JS object, useful for performance)
const enum HttpStatus {
    OK = 200,
    BAD_REQUEST = 400
}

let statusResult: HttpStatus = HttpStatus.OK;
// When compiled, `statusResult === 200` directly. No `HttpStatus` object exists at runtime.

// When to use Enums vs Union Types:
// Use enums when you have a distinct set of related constants that might need reverse mapping
// or when you want to group them semantically.
// Use union types (`"red" | "green" | "blue"`) when you have a limited set of string or number literals
// and don't need the extra features of enums (like reverse mapping or grouping constants).
// Union types are generally preferred for simplicity if the enum features aren't needed.
```

## 10. Type Assertions and Type Guards

TypeScript provides mechanisms to explicitly tell the compiler about a variable's type, or to narrow down a union type.

### Type Assertions
Type assertions are a way to tell the TypeScript compiler that you know more about the type of a value than it currently does. It's like a type cast in other languages, but it doesn't perform any runtime checks or data transformations.

```typescript
// 'as' syntax (preferred)
let someValue: any = "this is a string";
let strLength: number = (someValue as string).length;
console.log(strLength); // 16

// Angle-bracket syntax (not compatible with JSX)
let someOtherValue: any = "this is another string";
let anotherStrLength: number = (<string>someOtherValue).length;
console.log(anotherStrLength); // 20

// Type assertion for DOM elements
const myCanvas = document.getElementById("myCanvas") as HTMLCanvasElement;
if (myCanvas instanceof HTMLCanvasElement) { // Runtime check to be safe
    const ctx = myCanvas.getContext("2d");
    // ... use ctx
}

// Asserting a more specific type to a less specific type is usually fine
// Asserting a less specific type to a more specific type can be dangerous
// unless you assert to 'any' first.

// Example of potentially unsafe assertion
interface Bird {
    fly(): void;
    layEggs(): void;
}

interface Fish {
    swim(): void;
    layEggs(): void;
}

function getPet(): Bird | Fish {
    return {
        fly: () => console.log("flying"),
        layEggs: () => console.log("laying eggs")
    } as Bird; // Asserting that the returned object is a Bird
}

let pet = getPet();
// (pet as Bird).fly(); // This works, but TypeScript doesn't ensure `getPet` actually returns a Bird
```

### Type Guards
Type guards are conditional checks that narrow the type of a variable within a certain scope. TypeScript's control flow analysis uses these to provide more precise type information.

```typescript
// 1. 'typeof' Type Guard
function printId(id: string | number): void {
    if (typeof id === "string") {
        console.log(id.toUpperCase()); // id is 'string' here
    } else {
        console.log(id.toFixed(2)); // id is 'number' here
    }
}
printId("hello");
printId(123.456);

// 2. 'instanceof' Type Guard
class Dog {
    bark(): void { console.log("Woof!"); }
}
class Cat {
    meow(): void { console.log("Meow!"); }
}

function speak(animal: Dog | Cat): void {
    if (animal instanceof Dog) {
        animal.bark(); // animal is 'Dog' here
    } else {
        animal.meow(); // animal is 'Cat' here
    }
}
speak(new Dog());
speak(new Cat());

// 3. 'in' Operator Type Guard
interface Car {
    drive(): void;
}

interface Boat {
    sail(): void;
}

function navigate(vehicle: Car | Boat): void {
    if ("drive" in vehicle) {
        vehicle.drive(); // vehicle is 'Car' here
    } else {
        vehicle.sail(); // vehicle is 'Boat' here
    }
}

// 4. Discriminated Unions (using a common literal property)
interface Circle {
    kind: "circle"; // Discriminant property
    radius: number;
}

interface Square {
    kind: "square"; // Discriminant property
    sideLength: number;
}

type Shape = Circle | Square;

function getArea(shape: Shape): number {
    switch (shape.kind) {
        case "circle":
            return Math.PI * shape.radius ** 2; // shape is Circle
        case "square":
            return shape.sideLength ** 2;       // shape is Square
        default:
            // This 'never' check ensures all cases are handled.
            // If a new shape type is added to 'Shape' but not here,
            // TypeScript will show a compile-time error.
            const _exhaustiveCheck: never = shape;
            return _exhaustiveCheck;
    }
}

// 5. Custom Type Guards (User-Defined Type Guards)
// A function that returns a type predicate: 'parameterName is Type'
interface FishFood {
    swim(): void;
}

interface BirdFood {
    fly(): void;
}

function isFish(pet: FishFood | BirdFood): pet is FishFood {
    return (pet as FishFood).swim !== undefined; // Check for a unique property/method
}

function playWithPet(pet: FishFood | BirdFood): void {
    if (isFish(pet)) {
        pet.swim(); // pet is 'FishFood' here
    } else {
        pet.fly(); // pet is 'BirdFood' here
    }
}

playWithPet({ swim: () => console.log("Fish swimming") });
playWithPet({ fly: () => console.log("Bird flying") });
```

## 11. Generics

Generics are a way to make components reusable and type-safe. They allow you to write functions, classes, and interfaces that work with a variety of types, rather than just one.

```typescript
// Generic Function
// <T> is a type variable. It captures the type that the caller provides.
function identity<T>(arg: T): T {
    return arg;
}

// Using the generic function
let output1 = identity<string>("myString"); // output1 is type 'string'
let output2 = identity<number>(100);       // output2 is type 'number'
let output3 = identity(true);              // Type is inferred as 'boolean'

console.log(output1);
console.log(output2);
console.log(output3);

// Generic Function with multiple type variables
function swap<T, U>(a: T, b: U): [U, T] {
    return [b, a];
}

let swapped = swap(10, "hello"); // swapped is type '[string, number]'
console.log(swapped); // ["hello", 10]

// Generic with constraints (e.g., ensuring a type has a 'length' property)
interface Lengthwise {
    length: number;
}

function loggingIdentity<T extends Lengthwise>(arg: T): T {
    console.log(arg.length); // Now we know 'arg' has a .length property
    return arg;
}

loggingIdentity({ length: 10, value: 3 }); // OK
// loggingIdentity(3); // Error: Argument of type 'number' is not assignable to parameter of type 'Lengthwise'.

// Generic Classes
class GenericNumber<T> {
    zeroValue: T;
    add: (x: T, y: T) => T;

    constructor(zero: T, adder: (x: T, y: T) => T) {
        this.zeroValue = zero;
        this.add = adder;
    }
}

let myGenericNumber = new GenericNumber<number>(0, (x, y) => x + y);
console.log(myGenericNumber.add(5, 10)); // 15

let myGenericString = new GenericNumber<string>("", (x, y) => x + y);
console.log(myGenericString.add("Hello", "TypeScript")); // "HelloTypeScript"

// Generic Interfaces
interface KeyValuePair<K, V> {
    key: K;
    value: V;
}

let pair: KeyValuePair<string, number> = { key: "age", value: 30 };
console.log(pair.key, pair.value);

// Generic with default types
interface Box<T = string> { // Default type 'string' if not specified
    value: T;
}

let stringBox: Box = { value: "my string" }; // T is string
let numberBox: Box<number> = { value: 123 };  // T is number
```

## 12. DOM Manipulation

TypeScript integrates with the DOM by providing type definitions for all global DOM objects, methods, and properties. This means you get autocompletion and compile-time checks when interacting with the web page.

### Selecting Elements
```typescript
// Select by ID (TypeScript knows the general HTMLElement, often requires casting for specific elements)
let elementById = document.getElementById("myId"); // type: HTMLElement | null

// Type assertion to a more specific element type
const myDivElement = document.getElementById("myDiv") as HTMLDivElement;
if (myDivElement) {
    myDivElement.textContent = "I'm a div!";
}

// Select by class name
let elementsByClass = document.getElementsByClassName("myClass"); // type: HTMLCollectionOf<Element>
let firstByClass = elementsByClass[0]; // type: Element | undefined

// Select by tag name
```

## 13. Error Handling

TypeScript doesn't change how `try...catch` works at runtime, but it enhances compile-time safety when dealing with errors. Crucially, `catch` block variables are `unknown` by default in TypeScript 4.4+, requiring type narrowing.

### Try-Catch-Finally
```typescript
// Basic try-catch with unknown error type
try {
    let result = riskyOperation(); // Assume this function might throw
    console.log(result);
} catch (error: unknown) { // 'error' is 'unknown' by default in recent TS versions
    if (error instanceof Error) { // Type guard to narrow 'error' to 'Error'
        console.error("An error occurred:", error.message);
    } else {
        console.error("An unknown error occurred:", error);
    }
}

// Specific error types
function parseJsonSafe(jsonString: string): any {
    try {
        return JSON.parse(jsonString);
    } catch (error: unknown) {
        if (error instanceof SyntaxError) { // Narrow to SyntaxError
            console.log("JSON syntax error:", error.message);
        } else if (error instanceof ReferenceError) { // Narrow to ReferenceError
            console.log("Reference error:", error.message);
        } else if (error instanceof Error) { // Catch other Error types
            console.log("Generic error:", error.message);
        } else {
            console.log("Unknown error during parsing:", error);
        }
        throw error; // Re-throw if not handled gracefully
    }
}

parseJsonSafe("invalid json");
// parseJsonSafe("{\"key\": \"value\"}");

// Try-catch-finally
function processDataWithCleanup(jsonString: string): void {
    try {
        // Risky code
        const data = JSON.parse(jsonString);
        console.log("Data parsed:", data);
        // processData(data); // Assume processData exists
    } catch (error: unknown) {
        if (error instanceof Error) {
            console.error("Processing failed:", error.message);
        } else {
            console.error("An unexpected error occurred during processing.");
        }
    } finally {
        // Always executed
        console.log("Cleanup operations completed.");
    }
}
```


### Inheritance

### Encapsulation and Private Fields
TypeScript offers two ways for encapsulation:
1.  **`private` keyword**: TypeScript-specific, enforced at compile-time. Property is still accessible in compiled JavaScript.
2.  **`#` (Hash) Private Class Fields**: A new JavaScript feature (ES2022), enforced at runtime. TypeScript supports this.

```typescript
// Using TypeScript's 'private' keyword (compile-time encapsulation)
class BankAccountTS {
    private _balance: number = 0; // TypeScript 'private' property
    private readonly _accountNumber: string; // Readonly private property

    constructor(public owner: string, accountNumber: string) {
        this._accountNumber = accountNumber;
    }

    getBalance(): number {
        return this._balance;
    }

    deposit(amount: number): string {
        if (amount > 0) {
            this._balance += amount;
            return `Deposited ${amount}. New balance: ${this._balance}`;
        }
        throw new Error("Deposit amount must be positive");
    }

    withdraw(amount: number): string {
        if (amount > 0 && amount <= this._balance) {
            this._balance -= amount;
            return `Withdrew ${amount}. New balance: ${this._balance}`;
        }
        throw new new Error("Invalid withdrawal amount or insufficient balance");
    }
    
    // Attempting to access from outside will be a compile-time error
    // getAccountNumber(): string {
    //     return this._accountNumber;
    // }
}

console.log(accountTS.withdraw(500));
console.log(accountTS.getBalance());

    #accountNumber: string; // JS private field
    
    constructor(public owner: string, accountNumber: string) {
        this.#accountNumber = accountNumber;

        return this.#balance;
    }

            this.#balance += amount;
            return `Deposited ${amount}. New balance: ${this.#balance}`;
        }
        throw new Error("Deposit amount must be positive");
    }

    #validateTransaction(amount: number): boolean {
        return amount > 0 && amount <= this.#balance;

        if (this.#validateTransaction(amount)) { // Calling private method

        }
```

```typescript
// Get all keys, values, or entries
console.log([...myMap.keys()]);     // ["name", 1, true]
console.log([...myMap.values()]);   // ["John", "number key", "boolean key"]
console.log([...myMap.entries()]);  // [["name", "John"], [1, "number key"], [true, "boolean key"]]

// Convert Map to Object (keys must be string | number | symbol)
let mapToObj: { [key: string]: string | number | boolean | symbol } = Object.fromEntries(myMap);

// Convert Object to Map
```

## 18. Advanced Concepts

TypeScript augments JavaScript's advanced features by providing types and compile-time checks, making them safer and easier to reason about.

### Closures
Closures work exactly the same in TypeScript as in JavaScript. TypeScript's contribution is to provide type inference and checking for the variables captured in the closure.

```typescript
// Basic closure
function outerFunctionTyped(x: number): (y: number) => number {
    return function innerFunction(y: number): number {
        return x + y;  // Inner function has access to outer function's variable
    };
}

let addFive: (y: number) => number = outerFunctionTyped(5);
console.log(addFive(3));  // 8

// Practical closure example - counter with explicit return type for the object
function createCounterTyped(): {
    increment: () => number;
    decrement: () => number;
    getCount: () => number;
} {
    let count: number = 0; // Type inferred as number
    return {
        increment: () => ++count,
        decrement: () => --count,
        getCount: () => count
    };
}

let counterTyped = createCounterTyped();
console.log(counterTyped.increment());  // 1
console.log(counterTyped.increment());  // 2
console.log(counterTyped.getCount());   // 2

// Module pattern using closures (IIFE) with types
const CalculatorTyped = (function() {
    let result: number = 0; // Private state
    
    interface CalculatorAPI {
        add(x: number): CalculatorAPI;
        subtract(x: number): CalculatorAPI;
        multiply(x: number): CalculatorAPI;
        getResult(): number;
        reset(): CalculatorAPI;
    }

    const api: CalculatorAPI = {
        add: function(x: number) {
            result += x;
            return this;
        },
        subtract: function(x: number) {
            result -= x;
            return this;
        },
        multiply: function(x: number) {
            result *= x;
            return this;
        },
        getResult: function() {
            return result;
        },
        reset: function() {
            result = 0;
            return this;
        }
    };
    return api;
})();

// Method chaining
CalculatorTyped.add(10).multiply(2).subtract(5);
console.log(CalculatorTyped.getResult());  // 15
```

### Hoisting
Hoisting is a JavaScript behavior, and TypeScript's compiler understands it. While `var` and function declarations are hoisted, `let` and `const` variables are in a "temporal dead zone." TypeScript will catch `ReferenceError`s at compile time for `let`/`const`.

```typescript
// Variable hoisting (with 'var')
console.log(xVar);  // undefined (not an error)
var xVar: number = 5;

// This is what actually happens due to hoisting (in JS):
// var xVar;
// console.log(xVar);  // undefined
// xVar = 5;

// let and const are not hoisted in the same way (temporal dead zone)
// console.log(yLet);  // ReferenceError: Cannot access 'yLet' before initialization (TypeScript catches this)
let yLet: number = 10;

// Function hoisting
sayHelloTS();  // Works! Prints "Hello!" (Function declaration is hoisted)

function sayHelloTS(): void {
    console.log("Hello!");
}

// Function expressions are not hoisted
// sayGoodbyeTS();  // Error: Cannot access 'sayGoodbyeTS' before initialization (TypeScript catches this)
const sayGoodbyeTS = function(): void {
    console.log("Goodbye!");
};
```

### `this` Keyword
The behavior of `this` is a JavaScript runtime concept. TypeScript helps by providing contextually aware typing for `this`, especially within classes and with methods like `call`, `apply`, and `bind`.

```typescript
// Global context (in browser, 'this' is Window; in Node.js, 'this' is global object/module.exports)
// console.log(this); // In a module, `this` at the top-level is `undefined` in strict mode.

// Object method
let personThis = {
    name: "Alice",
    greet: function(this: typeof personThis): void { // Explicitly type 'this'
        console.log(`Hello, I'm ${this.name}`);
    },
    // Arrow function doesn't have its own 'this', it captures 'this' from the enclosing scope.
    // In this context, if 'personThis' is a global object, 'this' inside arrowMethod might be the global object.
    arrowMethod: () => {
        // console.log(this.name); // Error: 'this' implicitly has type 'any' because it does not have a type annotation.
                                 // Or, if 'noImplicitThis' is true, error: 'this' is not implicitly typed.
                                 // If compiled with es5/commonjs module, 'this' might refer to exports or global.
    }
};

personThis.greet();      // "Hello, I'm Alice"
// personThis.arrowMethod(); // Behaviour depends on compilation target and environment.

// Function context
function regularFunctionThis(this: Window | undefined): void { // Type 'this' to be explicit
    console.log(this);  // Global object (Window in browser, or undefined in strict mode)
}
regularFunctionThis();

// Constructor function (with type safety)
function PersonCtor(this: { name: string; greet: () => void; }, name: string): void {
    this.name = name;
    this.greet = function(this: { name: string; }): void { // Type 'this' explicitly here too
        console.log(`Hello, I'm ${this.name}`);
    };
}

let aliceThis = new (PersonCtor as any)("Alice"); // Cast if using direct function as constructor
aliceThis.greet();  // "Hello, I'm Alice"

// Explicit binding with call, apply, bind (TypeScript infers types correctly)
interface Nameable { name: string; }

function introduce(this: Nameable, greeting: string, punctuation: string): void {
    console.log(`${greeting}, I'm ${this.name}${punctuation}`);
}

let person1This: Nameable = { name: "Bob" };
let person2This: Nameable = { name: "Charlie" };

// call() - arguments passed individually
introduce.call(person1This, "Hi", "!");      // "Hi, I'm Bob!"

// apply() - arguments passed as array
introduce.apply(person2This, ["Hello", "."]);  // "Hello, I'm Charlie."

// bind() - creates new function with bound 'this'
let boundIntroduce = introduce.bind(person1This, "Hey"); // Partial application of arguments
boundIntroduce("!!!");  // "Hey, I'm Bob!!!"
```

### Generators (ES6)
TypeScript supports generator functions and iterable/iterator types, allowing you to define types for the yielded values and the return value.

```typescript
// Generator function with type annotations for yielded values and return value
function* numberGeneratorTyped(): Generator<number, string, undefined> {
    yield 1; // Yields a number
    yield 2;
    yield 3;
    return "Done"; // Returns a string when iteration finishes
}

let gen = numberGeneratorTyped();
console.log(gen.next());  // { value: 1, done: false }
console.log(gen.next());  // { value: 2, done: false }
console.log(gen.next());  // { value: 3, done: false }
console.log(gen.next());  // { value: "Done", done: true }

// Infinite generator
function* infiniteSequenceTyped(): Generator<number, void, undefined> {
    let i: number = 0;
    while (true) {
        yield i++;
    }
}

let infinite = infiniteSequenceTyped();
console.log(infinite.next().value);  // 0
console.log(infinite.next().value);  // 1
console.log(infinite.next().value);  // 2

// Generator with parameters (value sent back to generator using next())
function* parameterGeneratorTyped(): Generator<string, string, string> {
    let value: string = yield "First"; // Yields string, expects string from next()
    return `Received: ${value}`; // Returns string
}

let paramGen = parameterGeneratorTyped();
console.log(paramGen.next());           // { value: "First", done: false }
console.log(paramGen.next("Hello"));    // { value: "Received: Hello", done: true } (the generator finished)

// Fibonacci generator
function* fibonacciGeneratorTyped(): Generator<number, void, undefined> {
    let a: number = 0, b: number = 1;
    while (true) {
        yield a;
        [a, b] = [b, a + b];
    }
}

let fib = fibonacciGeneratorTyped();
for (let i = 0; i < 10; i++) {
    console.log(fib.next().value);
}

// Using generators with for...of (TypeScript understands iterable protocol)
function* rangeTyped(start: number, end: number): Generator<number, void, undefined> {
    for (let i = start; i <= end; i++) {
        yield i;
    }
}

for (let num of rangeTyped(1, 5)) {
    console.log(num);  // 1, 2, 3, 4, 5
}
```

### Symbols
TypeScript treats `symbol` as a primitive type and provides typings for well-known symbols.

```typescript
// Creating symbols
let sym1TS: symbol = Symbol();
let sym2TS: symbol = Symbol("description");
let sym3TS: symbol = Symbol("description");

console.log(sym2TS === sym3TS);  // false (each symbol is unique)
console.log(sym2TS.toString()); // "Symbol(description)"

// Symbols as object properties (TypeScript distinguishes symbol properties from string properties)
let objWithSymbols: {
    [key: string]: any; // String index signature
    [sym1TS]: string;   // Symbol index signature
    [ageSymbolTS]: number; // Explicit symbol property
} = {
    regularProperty: "regular value"
};

const nameSymbolTS: symbol = Symbol("name");
const ageSymbolTS: symbol = Symbol("age");

objWithSymbols[nameSymbolTS] = "John"; // Assigning to dynamic symbol property
objWithSymbols[ageSymbolTS] = 30; // Assigning to explicit symbol property

console.log(objWithSymbols[nameSymbolTS]);  // "John"
console.log(objWithSymbols[ageSymbolTS]);   // 30

// Symbols are not enumerable in for...in or Object.keys()

    console.log(key);  // Only "regularProperty"
}

console.log(Object.keys(objWithSymbols));           // ["regularProperty"]
console.log(Object.getOwnPropertySymbols(objWithSymbols));  // [Symbol(name), Symbol(age)]

// Well-known symbols (TypeScript has definitions for these)
let myArrayIterator: number[] = [1, 2, 3];
// Override default iterator for an array to reverse it
myArrayIterator[Symbol.iterator] = function* (): Generator<number, void, undefined> {
    for (let i = this.length - 1; i >= 0; i--) {
        yield this[i];
    }
};

for (let item of myArrayIterator) {
    console.log(item);  // 3, 2, 1 (reversed)
}

// Symbol.for() - global symbol registry

console.log(Symbol.keyFor(globalSym1));  // "global"
```

### Iterators and Iterables
TypeScript provides interfaces (`Iterable`, `Iterator`, `IterableIterator`) to work with custom iterators and iterables, ensuring type safety when you implement or consume them.

```typescript
// Creating a custom iterable with explicit types
let myIterableTyped: Iterable<number> = {
    data: [1, 2, 3, 4, 5],
    [Symbol.iterator]: function(): Iterator<number> {
        let index: number = 0;
        let data: number[] = this.data as number[]; // Assert type for 'this.data'

        return {
            next: function(): IteratorResult<number> {
                if (index < data.length) {
                    return { value: data[index++], done: false };
                } else {
                    return { done: true, value: undefined }; // Value is optional for done:true
                }
            }
        };
    }
};

// Using the iterable (TypeScript understands the protocol)
for (let value of myIterableTyped) {
    console.log(value);  // 1, 2, 3, 4, 5
}

// Manual iteration
let iterator: Iterator<number> = myIterableTyped[Symbol.iterator]();
console.log(iterator.next());  // { value: 1, done: false }
console.log(iterator.next());  // { value: 2, done: false }

// Range iterable using a generator (simpler to implement iterables)
function rangeIterable(start: number, end: number): Iterable<number> {
    return {
        [Symbol.iterator]: function* (): Generator<number> { // Returns a Generator (which is an IterableIterator)
            for (let i = start; i <= end; i++) {
                yield i;
            }
        }
    };
}

for (let num of rangeIterable(1, 5)) {
    console.log(num);  // 1, 2, 3, 4, 5
}

// Spread operator with iterables (TypeScript correctly infers the resulting array type)
console.log([...rangeIterable(1, 3)]);  // [1, 2, 3]
```

### Arrow Functions
```typescript

let sumArrow: number = numbersArrow.reduce((acc, x) => acc + x, 0);

// This is a key difference and often a reason to use them in callbacks.
let objArrow = {
    name: "Alice",

        console.log(this.name);  // "Alice" (this refers to objArrow)

        let arrowFunctionInsideRegular = () => {
            console.log(this.name);  // "Alice" (lexically captures 'this' from regularMethod)
        };
        arrowFunctionInsideRegular();
    },

    arrowMethod: () => {
        // console.log(this.name); // 'this' here refers to the 'this' of the scope where objArrow is defined (e.g., global/module scope)
                                // If 'noImplicitThis' is true, TS will complain unless 'this' is explicitly typed or globally defined.
    }
};

objArrow.regularMethod();

// Arrow functions can't be used as constructors
// const PersonArrow = (name: string) => { this.name = name; };
// const personArrow = new PersonArrow("Bob"); // TypeError: PersonArrow is not a constructor

// Arrow functions don't have arguments object
function regularFunctionArguments(a: number, b: number): void {
    console.log(arguments);  // Arguments object (e.g., { '0': 1, '1': 2 })
}
regularFunctionArguments(1, 2);

const arrowFunctionArguments = (...args: number[]): void => { // Use rest parameters instead
    console.log(args);  // Array of arguments (e.g., [1, 2])
};
arrowFunctionArguments(1, 2);
```

```typescript
interface UserPage {

async function* fetchUserPages(): AsyncGenerator<UserAPI[], void, undefined> {
    let page: number = 1;
    let hasMore: boolean = true;
    while (hasMore) {
        const response: Response = await fetch(`/api/users?page=${page}`);
        if (!response.ok) throw new Error(`Failed to fetch page ${page}: ${response.status}`);
        const data: UserPage = await response.json();

        if (data.users.length === 0) {
            hasMore = false;
        } else {
            yield data.users;
            page++;
            if (!data.nextPage) hasMore = false; // Assume no nextPage means no more pages
        }
    }
}
```

```typescript
// Debouncing for frequent events (e.g., search input, resize)
function debounce<T extends (...args: any[]) => any>(func: T, wait: number): (...args: Parameters<T>) => void {
    let timeout: ReturnType<typeof setTimeout>;
    return function executedFunction(...args: Parameters<T>) {
        const later = () => {
            clearTimeout(timeout);
            func(...args);
        };
        clearTimeout(timeout);
        timeout = setTimeout(later, wait);
    };
}

// Throttling for scroll events, mousemove etc.
function throttle<T extends (...args: any[]) => any>(func: T, limit: number): (...args: Parameters<T>) => void {
    let inThrottle: boolean;
    let lastFn: ReturnType<typeof setTimeout>;
    let lastTime: number;
    return function throttledFunction(this: any, ...args: Parameters<T>) {
        const context = this;
        if (!inThrottle) {
            func.apply(context, args);
            lastTime = Date.now();
            inThrottle = true;
        } else {
            clearTimeout(lastFn);
            lastFn = setTimeout(() => {
                if (Date.now() - lastTime >= limit) {
                    func.apply(context, args);
                    lastTime = Date.now();
                }
            }, Math.max(limit - (Date.now() - lastTime), 0));
        }
    };
}
```

```typescript
// api-client.ts - Handles API interactions
export class APIClient {
    private baseUrl: string;

    constructor(baseURL: string) {
        this.baseUrl = baseURL;
    }

    async get<T>(endpoint: string): Promise<T> {
        const response = await fetch(`${this.baseUrl}${endpoint}`);
        return this.handleResponse<T>(response);
    }

    async post<T>(endpoint: string, data: any): Promise<T> {
        const response = await fetch(`${this.baseUrl}${endpoint}`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(data)
        });
        return this.handleResponse<T>(response);
    }

    private async handleResponse<T>(response: Response): Promise<T> {
        if (!response.ok) {
            const errorText = await response.text();
            throw new Error(`API Error ${response.status}: ${errorText}`);
        }
        return response.json() as Promise<T>;
    }
}

export function generateUUID(): string {
    return 'xxxxxxxx-xxxx-4xxx-yxxx-xxxxxxxxxxxx'.replace(/[xy]/g, function(c) {
        var r = Math.random() * 16 | 0, v = c == 'x' ? r : (r & 0x3 | 0x8);
        return v.toString(16);
    });
}
```
